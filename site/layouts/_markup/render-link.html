{{- /* Markdown links are written as relative paths to files in the repository, so they resolve on GitHub and pass the link checks. This hook maps each one to the site page that renders that file, and sends everything else to GitHub. */ -}} {{- $u := .Destination -}} {{- $href := $u -}} {{- $external := or (hasPrefix $u "http://") (hasPrefix $u "https://") (hasPrefix $u "mailto:") -}} {{- if and (not $external) (not (hasPrefix $u "#")) (not (hasPrefix $u "/")) -}} {{- $parts := split $u "#" -}} {{- $file := index $parts 0 -}} {{- $frag := "" -}} {{- if gt (len $parts) 1 }}{{ $frag = printf "#%s" (index $parts 1) }}{{ end -}} {{- /* The directory of the source file, relative to the repository root. */ -}} {{- $dir := "" -}} {{- with .Page.Params.sourcedir -}} {{- $dir = . -}} {{- else -}} {{- with .Page.File }}{{ $dir = path.Join "site/content" .Dir }}{{ end -}} {{- end -}} {{- if eq $dir "." }}{{ $dir = "" }}{{ end -}} {{- $full := path.Clean (path.Join $dir $file) -}} {{- $pages := dict "docs/ENGINE.md" "/docs/engine/" "docs/CLI.md" "/docs/cli/" "docs/PROVIDERS.md" "/docs/providers/" "docs/PERSONAS.md" "/docs/personas/" "docs/COVERAGE.md" "/docs/coverage/" "docs/ARCHITECTURE.md" "/docs/architecture/" "docs/ETHICS.md" "/docs/ethics/" "docs/TELEMETRY.md" "/docs/telemetry/" "docs/PLAN.md" "/docs/plan/" "docs/LICENSING.md" "/docs/licensing/" "docs/STYLE.md" "/docs/style/" "docs/TYPOGRAPHY.md" "/docs/typography/" "docs/NAMING.md" "/docs/naming/" "docs/integrations/grafana.md" "/docs/grafana-guide/" "CONTRIBUTING.md" "/docs/contributing/" "SECURITY.md" "/docs/security/" "TRADEMARKS.md" "/docs/trademarks/" "brand/README.md" "/docs/brand/" "README.md" "/" -}} {{- with index $pages $full -}} {{- $href = printf "%s%s" . $frag -}} {{- else -}} {{- if and (hasPrefix $full "site/content/") (hasSuffix $full ".md") -}} {{- /* A page written for the site: site/content/for/developer.md is /for/developer/. */ -}} {{- $rel := strings.TrimSuffix ".md" (strings.TrimPrefix "site/content/" $full) -}} {{- $rel = strings.TrimSuffix "_index" $rel -}} {{- $rel = strings.TrimSuffix "/" $rel -}} {{- if $rel }}{{ $href = printf "/%s/%s" $rel $frag }}{{ else }}{{ $href = printf "/%s" $frag }}{{ end -}} {{- else -}} {{- $kind := "blob" -}} {{- if or (hasSuffix $file "/") (not (path.Ext $file)) }}{{ $kind = "tree" }}{{ end -}} {{- $href = printf "%s/%s/main/%s%s" site.Params.repo $kind $full $frag -}} {{- end -}} {{- end -}} {{- end -}} {{ .Text }} {{- /* chomp */ -}}

{{- /* An image in a page rendered from the repository's Markdown is written as a relative path, so it resolves on GitHub. This hook finds the file among the mounted repository docs and publishes it with the site. Anything else is written as it is. */ -}} {{- $src := .Destination -}} {{- $res := "" -}} {{- with .Page.Params.sourcedir -}} {{- if not (or (hasPrefix $src "http") (hasPrefix $src "/")) -}} {{- $res = resources.Get (path.Join "repo" . $src) -}} {{- end -}} {{- end -}} {{- if $res -}} {{- else -}} {{- end -}}
